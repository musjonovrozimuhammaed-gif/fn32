#include <stdio.h>

int main() {
    int tanlov;
    
    printf("Transport turini tanlang:\n");
    printf("1 - Avtomobil\n");
    printf("2 - Avtobus\n");
    printf("3 - Poyezd\n");
    printf("4 - Samolyot\n");
    printf("Raqamni kiriting (1-4): ");
    scanf("%d", &tanlov);
    
    switch (tanlov) {
        case 1:
            printf("Siz **Avtomobil**ni tanladingiz.\n");
            break;
        case 2:
            printf("Siz **Avtobus**ni tanladingiz.\n");
            break;
        case 3:
            printf("Siz **Poyezd**ni tanladingiz.\n");
            break;
        case 4:
            printf("Siz **Samolyot**ni tanladingiz.\n");
            break;
        default:
            printf("Noto'g'ri raqam kiritdingiz! Iltimos, 1 dan 4 gacha bo'lgan raqamlardan birini tanlang.\n");
            break;
    }
    
    return 0;
}
