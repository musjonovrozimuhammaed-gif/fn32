    int jami_pul = 0;
    
    printf("Jamg'arma dasturiga xush kelibsiz! Maqsad: 100 000 so'm.\n");
    
    
    while (jami_pul < 100000) {
        printf("Qancha pul qo'shdingiz? ");

        jami_pul += qoshimcha_pul;

    }
    
    printf("Tabriklaymiz! Maqsadga erishildi.\n");

    return 0;
}
