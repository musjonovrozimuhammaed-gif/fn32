#include <stdio.h>

int main() {
    int tanlov;
    double som, natija;
    double kurs_usd = 12800.0;
    double kurs_eur = 13500.0;
    double kurs_rub = 130.0;
    
    printf("Valyuta turini tanlang:\n");
    printf("1 - Dollar (USD)\n");
    printf("2 - Yevro (EUR)\n");
    printf("3 - Rubl (RUB)\n");
    printf("Raqamni kiriting (1-3): ");
    scanf("%d", &tanlov);
    
    printf("Almashtirmoqchi bo'lgan so'm miqdorini kiriting: ");
    scanf("%lf", &som);
    
    switch (tanlov) {
        case 1:
            natija = som / kurs_usd;
            printf("%.2f so'm = %.2f AQSh dollari (USD)\n", som, natija);
            break;
        case 2:
            natija = som / kurs_eur;
            printf("%.2f so'm = %.2f Yevro (EUR)\n", som, natija);
            break;
        case 3:
            natija = som / kurs_rub;
            printf("%.2f so'm = %.2f Rossiya rubli (RUB)\n", som, natija);
            break;
        default:
            printf("Xato! Noto'g'ri valyuta raqamini tanladingiz.\n");
            break;
    }
    
    return 0;
}
